cmake_minimum_required(VERSION 3.25)
project(BowSpeedVR VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/EHsc /W4)
    # Cover dependency source_location strings as well as our own __FILE__ paths.
    # MSVC ignores /pathmap without /experimental:deterministic.
    add_compile_options(/experimental:deterministic)
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" BOWSPEED_PUBLIC_ROOT)
    add_compile_options("/pathmap:${BOWSPEED_PUBLIC_ROOT}=BowSpeedVR")
    if(COMMONLIB_SOURCE_DIR)
        file(TO_NATIVE_PATH "${COMMONLIB_SOURCE_DIR}" BOWSPEED_PUBLIC_COMMONLIB)
        add_compile_options("/pathmap:${BOWSPEED_PUBLIC_COMMONLIB}=CommonLibSSE-NG")
    endif()
    if(VCPKG_INSTALLED_DIR)
        file(TO_NATIVE_PATH "${VCPKG_INSTALLED_DIR}" BOWSPEED_PUBLIC_DEPS)
        add_compile_options("/pathmap:${BOWSPEED_PUBLIC_DEPS}=Dependencies")
    endif()
endif()
option(BOWSPEED_BUILD_PLUGIN "Build the SKSEVR DLL" OFF)
add_library(bow_speed_core STATIC src/Config.cpp src/MatchedPhysics.cpp src/PhysicsCatalog.cpp src/MatchedRuntime.cpp src/UserConfig.cpp)
target_include_directories(bow_speed_core PUBLIC include)
enable_testing()
add_executable(bow_speed_tests tests/PolicyTests.cpp tests/ConfigTests.cpp tests/GuardTests.cpp tests/DiagnosticTests.cpp tests/AmmoPhysicsTests.cpp tests/ImpactThunkTests.cpp)
target_link_libraries(bow_speed_tests PRIVATE bow_speed_core)
add_test(NAME bow_speed_behavior COMMAND bow_speed_tests)
add_test(NAME bow_speed_function_guards COMMAND bow_speed_tests --guards)
add_test(NAME bow_speed_diagnostics COMMAND bow_speed_tests --diagnostics)
add_test(NAME bow_speed_ammo_physics COMMAND bow_speed_tests --ammo)
add_executable(bow_speed_matched_tests tests/MatchedPhysicsTests.cpp)
target_link_libraries(bow_speed_matched_tests PRIVATE bow_speed_core)
add_test(NAME bow_speed_matched_physics COMMAND bow_speed_matched_tests)
add_executable(bow_speed_catalog_tests tests/PhysicsCatalogTests.cpp)
target_link_libraries(bow_speed_catalog_tests PRIVATE bow_speed_core)
add_test(NAME bow_speed_physics_catalog COMMAND bow_speed_catalog_tests "${CMAKE_CURRENT_SOURCE_DIR}/config/BowSpeedVR-pairing.ini")
add_executable(bow_speed_runtime_tests tests/MatchedRuntimeTests.cpp)
target_link_libraries(bow_speed_runtime_tests PRIVATE bow_speed_core)
add_test(NAME bow_speed_matched_runtime COMMAND bow_speed_runtime_tests)
add_executable(bow_speed_tuning_tests tests/TuningTests.cpp)
target_link_libraries(bow_speed_tuning_tests PRIVATE bow_speed_core)
add_test(NAME bow_speed_standalone_tuning COMMAND bow_speed_tuning_tests "${CMAKE_CURRENT_SOURCE_DIR}/dist/SKSE/Plugins/BowSpeedVR.ini")
add_executable(bow_speed_inventory_tests tests/InventoryWeightTests.cpp)
target_link_libraries(bow_speed_inventory_tests PRIVATE bow_speed_core)
add_test(NAME bow_speed_inventory_weight COMMAND bow_speed_inventory_tests)
add_executable(bow_speed_user_tests tests/UserConfigTests.cpp)
target_link_libraries(bow_speed_user_tests PRIVATE bow_speed_core)
add_test(NAME bow_speed_user_config COMMAND bow_speed_user_tests)
if(BOWSPEED_BUILD_PLUGIN)
    if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "SKSEVR requires Windows x64")
    endif()
    if(NOT COMMONLIB_SOURCE_DIR)
        message(FATAL_ERROR "Set COMMONLIB_SOURCE_DIR to CommonLibSSE-NG 39f9d07a6ffabea8fb559eee87ab7d27cd463e8a")
    endif()
    set(BUILD_TESTS OFF CACHE BOOL "Disable dependency tests" FORCE)
    set(SKSE_SUPPORT_PATCH_SAFETY OFF CACHE BOOL "This plugin checks exact VR call-site bytes before patching" FORCE)
    set(COMMONLIB_ENABLE_IPO OFF CACHE BOOL "Faster reproducible source build" FORCE)
    add_subdirectory("${COMMONLIB_SOURCE_DIR}" commonlib)
    include("${COMMONLIB_SOURCE_DIR}/cmake/CommonLibSSE.cmake")
    add_commonlibsse_plugin(BowSpeedVR
        NAME "BowSpeedVR" AUTHOR "BowSpeedVR contributors"
        COMPATIBLE_RUNTIMES "1.4.15.0" STRUCT_DEPENDENT
        SOURCES src/Plugin.cpp src/SpeedHook.cpp src/ImpactHook.cpp src/InventoryWeightHook.cpp)
    target_link_libraries(BowSpeedVR PRIVATE bow_speed_core)
    target_compile_definitions(BowSpeedVR PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_precompile_headers(BowSpeedVR PRIVATE include/BowSpeed/PCH.h)
endif()
